-- Eliminar tabla si existe --
DROP TABLE IF EXISTS companies;

-- Crear tabla --
CREATE TABLE companies (
  ID INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL,
  address TEXT NOT NULL,
  city TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE,
  is_active INTEGER DEFAULT TRUE
);
